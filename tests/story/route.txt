# Story route for tests/story_compare.py (the port and the real game play it side by side).
# Steps: see src/story.h. Walks are planned with tools/plan.py on the port's live map
# (build/story/grid.txt, written where a --story run ends).

== opening
new                                   # title, names, the simulator battle, the prologue
check after_opening                   # aboard the Talon (map 0B 26,7)
battles off                           # random battles differ between the two (and can wipe the
                                      # party): the route is about places and story; battles have
                                      # their own tests
npcs still                            # wandering uses the real RNG (hFF93 frame count): not matchable

== leave the Talon
go U22                                # up the deck (the lift carries you) to the nose
check off_talon                       # the temple top by Dharm (map 0A), the Talon a swirl
go D12                                # down the stairs: the step text takes you outside
check overworld                       # map 02 32,30

== into Dharm
go D1 R1 D2 R3 U1 R2 U2 R2            # round the trees to the town gate
check dharm
go D1                                 # straight back out of the gate
check out_of_dharm

== to the North Tower
go L1 U7 L5 U8 R2 U1 R4 U7 L3 U1 L6 D4 R3  # north round the mountains to the tower door (13,32)
check north_tower

== North Tower, floor 1 to 2
go U12                                # up the corridor and the room to the stairs (5,52)
check tower_2f                        # up a floor: 10,30
go U1 R1
jump                                  # B: jump the pitfall at 9,32 (falling in drops you a floor)
check over_the_pit
go R4 U1 R1 U3                        # to the stairs at 5,38
check tower_map33

== North Tower, floor 3: Float
go U1 L2 D7 L9 U2 R1
jump
go U3                                 # the stairs warp to 28,58
go U4
face R
a                                     # the chest: Float
check float

== learn Float (Item: mark it, A again, pick the leader)
keys S w40 A w30 D w10 A w30 A w30 A w30
keys A w30 B w30 B w30
check menu_closed

== down and out of the North Tower
go D5
go D3                                 # a pitfall: back to floor 2 (map 32)
check back_on_32
go D1 R1 D2 R2
go D1 L5 D5
check outside_tower                   # map 02 14,32

== cast Float (Magic: the first A marks it, the second casts; 01:4049)
keys S w40 D w10 A w30 A w30 A w30
peek spell_list
keys A w120
peek floating                         # vehicle 1: 8 px up, bobbing, a shadow

== over the mountains to the town (map 20) (afloat: no step events, 00:0906)
go L1 U1 L4 D14 L2 D7 L12 U5 L1
check over_the_town
go D1
settle
keys B w30                            # B: off the float on plain ground (01:4311)
check landed
go U1
check town                            # map 20: canals and an inn

== through the inn (map 23) and its upper floor (map 22)
go D9 R2 D2 R1 D2 R1 D3 R4 U1
check inn_23
go U6 L2 D2
check floor_22
go D2 L2 U7 L5
jump
go D4 L9                              # over the cure floor ($E1)
check canal_walkway    # back in map 20, up on the walkway (flags $45)

== into the round tower (map 25)
go D1 L4 D2 L2
face U                                # onto the doorway object: it doesn't block (row bit 6, 01:465F)
check tower_door
go U1
check tower_25

== Granny: her speech, then her spell shop (left with B)
go U1
face U
shop
check granny

== to Cronos, in the largest house (map 21)
go D2
check out_of_tower
go D1 R2 U3 R5                        # the walkway's C5: a return - back up the stack to map 22
check back_on_22                      # known: bottom halves behind BG on the walkway (layouts, 00:3A7F)
go U2 R2 U1 R2 U1 R4 U1 R2 D1 R5 D7 R2 U3
check back_in_inn
go U1 R2 D7
check main_street
go D3 L1 D2 L1 D4 L1 D1 L5 U3
check cronos_house
go U9
face U
a
check cronos                          # takes the Radar, gives Past

== back to the Talon with Past (01:79B3: a Unit aboard lets it leave)
go D11
check out_of_house
go D2 L4 U22 R3 U5
check out_of_town
keys S w40 D w10 A w30 A w30 A w30
keys A w120
check afloat_again

settle

check below_temple
face U

check after_boss                      # lost - everyone on 1 HP, flag 4F cleared, the monster gone

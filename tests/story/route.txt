# Steps: see src/story.h. Walks are planned with tools/plan.py on the port's live map
# (build/story/grid.txt, written where a --story run ends).

== opening
new                                   # title, names, the simulator battle, the prologue
check after_opening                   # aboard the Talon (map 0B 26,7)

== leave the Talon
go U22                                # up the deck (the lift carries you) to the nose
check off_talon                       # the temple top by Dharm (map 0A), the Talon a swirl
go D12                                # down the stairs: the step text takes you outside
check overworld                       # map 02 32,30

== into Dharm
go D1 R1 D2 R3 U1 R2 U2 R2            # round the trees to the town gate
check dharm
go D1                                 # straight back out of the gate
check out_of_dharm

== to the North Tower
go L1 U7 L5 U8 R2 U1 R4 U7 L3 U1 L6 D4 R3  # north round the mountains to the tower door (13,32)
check north_tower

== North Tower, floor 1 to 2
go U12                                # up the corridor and the room to the stairs (5,52)
check tower_2f                        # up a floor: 10,30
go U1 R1
jump                                  # B: jump the pitfall at 9,32 (falling in drops you a floor)
check over_the_pit
go R4 U1 R1 U3                        # to the stairs at 5,38
check tower_map33

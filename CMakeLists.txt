cmake_minimum_required(VERSION 3.16)
project(ffl3 CXX C)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

# SDL2: use a system install if present, otherwise build the bundled copy in third_party/SDL2
# (git clone -b release-2.30.x https://github.com/libsdl-org/SDL.git third_party/SDL2)
find_package(SDL2 QUIET)
if(NOT SDL2_FOUND)
    if(NOT EXISTS ${CMAKE_SOURCE_DIR}/third_party/SDL2/CMakeLists.txt)
        include(FetchContent)
        FetchContent_Declare(SDL2 GIT_REPOSITORY https://github.com/libsdl-org/SDL.git
                                  GIT_TAG release-2.30.x GIT_SHALLOW TRUE)
        set(SDL_SHARED OFF CACHE BOOL "" FORCE)
        set(SDL_STATIC ON CACHE BOOL "" FORCE)
        set(SDL_TEST OFF CACHE BOOL "" FORCE)
        FetchContent_MakeAvailable(SDL2)
    else()
        set(SDL_SHARED OFF CACHE BOOL "" FORCE)
        set(SDL_STATIC ON CACHE BOOL "" FORCE)
        set(SDL_TEST OFF CACHE BOOL "" FORCE)
        add_subdirectory(third_party/SDL2 EXCLUDE_FROM_ALL)
    endif()
endif()

add_library(ffl3core STATIC src/rom.cpp src/mapdata.cpp src/field.cpp src/render.cpp src/text.cpp src/battledata.cpp src/battle.cpp src/battlevm.cpp src/battlenatives.cpp src/shop.cpp src/menu.cpp src/sound.cpp src/console.cpp src/uitext.cpp)
target_include_directories(ffl3core PUBLIC src)

add_executable(ffl3 WIN32 src/main.cpp)
target_link_libraries(ffl3 PRIVATE ffl3core)
if(TARGET SDL2::SDL2main)
    target_link_libraries(ffl3 PRIVATE SDL2::SDL2main)
endif()
if(TARGET SDL2::SDL2-static)
    target_link_libraries(ffl3 PRIVATE SDL2::SDL2-static)
else()
    target_link_libraries(ffl3 PRIVATE SDL2::SDL2)
endif()

if(MSVC)
    target_compile_options(ffl3core PRIVATE /W4)
else()
    target_compile_options(ffl3core PRIVATE -Wall -Wextra)
endif()

add_executable(vmreplay tests/vmreplay.cpp)
target_link_libraries(vmreplay PRIVATE ffl3core)

add_executable(sndreplay tests/sndreplay.cpp)
target_link_libraries(sndreplay PRIVATE ffl3core)
